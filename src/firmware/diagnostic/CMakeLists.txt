cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED ENV{IDF_PATH})
    message(FATAL_ERROR "Use existing ESP-IDF v5.5.6 environment. No installation or flashing is performed.")
endif()
set(IDF_TARGET "esp32s3" CACHE STRING "Readonly preparation target")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../components")
set(COMPONENTS main robot_diagnostic robot_twai robot_can robot_core)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(robot_arm_readonly)
if(NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "Readonly preparation is for ESP32-S3 only")
endif()
